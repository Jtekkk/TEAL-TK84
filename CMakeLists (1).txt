cmake_minimum_required(VERSION 3.22)

project(TEAL_TK84 VERSION 1.0.0)

# JUCE path - user should set this to their JUCE installation
# Example: set(JUCE_DIR "C:/JUCE")
if(NOT DEFINED JUCE_DIR)
    message(FATAL_ERROR "Please set JUCE_DIR to your JUCE installation path")
endif()

add_subdirectory(${JUCE_DIR} JUCE)

juce_add_plugin(TEAL_TK84
    COMPANY_NAME "TEAL"
    BUNDLE_ID "com.teal.tk84"
    PLUGIN_MANUFACTURER_CODE "Teal"
    PLUGIN_CODE "TK84"
    FORMATS VST3 AU AAX Standalone
    PRODUCT_NAME "TEAL TK84"
    DESCRIPTION "Stereo VCA Bus Compressor"
    IS_SYNTH FALSE
    NEEDS_MIDI_INPUT FALSE
    NEEDS_MIDI_OUTPUT FALSE
    IS_MIDI_EFFECT FALSE
    EDITOR_WANTS_KEYBOARD_FOCUS FALSE
    COPY_PLUGIN_AFTER_BUILD TRUE
)

target_sources(TEAL_TK84 PRIVATE
    PluginProcessor.cpp
    PluginEditor.cpp
    TK84Engine.cpp
    TK84Detector.cpp
    TK84GainComputer.cpp
    TK84VCA.cpp
    TK84Transient.cpp
    SteppedRotarySlider.cpp
    GRLEDMeter.cpp
    IlluminatedVUMeter.cpp
    TK84LookAndFeel.cpp
)

target_include_directories(TEAL_TK84 PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}
)

target_compile_definitions(TEAL_TK84 PUBLIC
    JUCE_WEB_BROWSER=0
    JUCE_USE_CURL=0
    JUCE_VST3_CAN_REPLACE_VST2=0
)

target_link_libraries(TEAL_TK84 PRIVATE
    juce::juce_audio_utils
    juce::juce_audio_processors
    juce::juce_gui_basics
    juce::juce_gui_extra
)
